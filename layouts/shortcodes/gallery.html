{{- /* gallery.html — {{< gallery folder="_img/xyz" exclude="a.jpg,b.jpg" >}} Grid of photos from a content folder with a lightbox. The originals are phone JPGs (4096px, 5–10 MB each); serving them as 180-px grid thumbs made the four gallery notes the heaviest pages on the site (Ahrefs "image file size too large", PSI "improve image delivery"). Now: grid thumb : 480×360 WebP q80 (+ 960×720 for 2× screens), ~25–40 KB lightbox : 2000-px-wide WebP q85, ~300–500 KB, still crisp full-screen The original file stays in place, untouched, just no longer linked. Files Hugo cannot process (gif, svg, anything not reachable as a resource) fall back to the plain original as before. jQuery + lightbox2 are deferred so they no longer block parsing on these pages. */ -}} {{- $folder := .Get "folder" -}} {{- $excludeStr := default "" (.Get "exclude") -}} {{- $exclude := split $excludeStr "," -}} {{- $imageExts := slice ".jpg" ".jpeg" ".png" ".gif" ".webp" -}} {{- $resizable := slice ".jpg" ".jpeg" ".png" ".webp" -}} {{- if not (.Page.Scratch.Get "lightbox2_loaded") -}} {{- .Page.Scratch.Set "lightbox2_loaded" true -}} {{- end -}}
{{- range readDir (printf "content/%s" $folder) -}} {{- if .IsDir -}}{{- continue -}}{{- end -}} {{- $name := .Name -}} {{- $ext := strings.ToLower (path.Ext $name) -}} {{- if and (in $imageExts $ext) (not (in $exclude $name)) -}} {{- $imgPath := printf "%s/%s" $folder $name -}} {{- $imgURL := $imgPath | relURL -}} {{- $title := $name | strings.TrimSuffix $ext -}} {{- $res := "" -}} {{- if in $resizable $ext -}}{{- $res = resources.Get (printf "notes/%s" $imgPath) -}}{{- end -}} {{- if and $res (gt $res.Width 0) -}} {{- $thumb := $res.Fill "480x360 webp q80 Lanczos" -}} {{- $thumb2x := $res.Fill "960x720 webp q80 Lanczos" -}} {{- /* if/else, not cond: cond evaluates both branches, which would upscale small images */ -}} {{- $full := $res -}} {{- if gt $res.Width 2000 -}}{{- $full = $res.Resize "2000x webp q85 Lanczos" -}}{{- end -}} {{- else -}} {{- end -}} {{- end -}} {{- end -}}
