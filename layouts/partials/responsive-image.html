{{- /* Renders one note image as a responsive . dict: path (string, the wikilink / markdown target, e.g. "Note-123.webp" or "_img/x.png"), width (string, optional: "400" or "353x116"), alt (string, optional), eager (bool: first image of the note). The vault images are copied flat into content/ and published as-is; config.toml additionally mounts content/ under assets/notes so they are also reachable as Hugo resources. For raster images wider than 800px we emit resized WebP variants (800w, 1200w, 1600w) plus the original, with width/height attributes so the browser reserves the space (no CLS). Anything that cannot be processed (svg, gif, missing file) falls back to the plain as before. */ -}} {{- $path := .path -}} {{- $widthAttr := .width | default "" -}} {{- $alt := .alt | default "" -}} {{- $eager := .eager | default false -}} {{- $src := $path | relURL -}} {{- $srcEnc := replace $src " " "%20" -}} {{- $ext := lower (path.Ext $path) -}} {{- $res := "" -}} {{- if in (slice ".png" ".jpg" ".jpeg" ".webp") $ext -}} {{- $res = resources.Get (printf "notes/%s" (strings.TrimPrefix "/" $path)) -}} {{- end -}} {{- $loading := cond $eager "eager" "lazy" -}} {{- $prio := cond $eager ` fetchpriority="high"` "" -}} {{- /* explicit width from `![[img|400]]` or `![[img|353x116]]` */ -}} {{- $reqW := 0 -}} {{- if findRE `^\d+(x\d+)?$` $widthAttr -}}{{- $reqW = int (index (split $widthAttr "x") 0) -}}{{- end -}} {{- if and $res (gt $res.Width 0) -}} {{- $w := $res.Width -}}{{- $h := $res.Height -}} {{- $dispW := cond (gt $reqW 0) $reqW $w -}} {{- $dispH := int (math.Round (mul $h (div (float $dispW) (float $w)))) -}} {{- $cands := slice -}} {{- range $bp := slice 800 1200 1600 -}} {{- if lt $bp $w -}} {{- $v := $res.Resize (printf "%dx webp q80" $bp) -}} {{- $cands = $cands | append (printf "%s %dw" (replace $v.RelPermalink " " "%20") $v.Width) -}} {{- end -}} {{- end -}} {{- $cands = $cands | append (printf "%s %dw" $srcEnc $w) -}} {{- $sizes := cond (gt $reqW 0) (printf "%dpx" $reqW) "(max-width: 900px) 100vw, 840px" -}} {{- else -}} {{- end -}}
